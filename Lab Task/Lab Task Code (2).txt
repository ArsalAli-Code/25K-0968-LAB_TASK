#include <iostream>
using namespace std;

class Employee{
public:
    int id;
    double salary;

    void showEmployee(){
        cout<<"ID: "<<id<<endl;
        cout<<"Salary: "<<salary<<endl;
    }
};

class Manager : public Employee{
public:
    double bonus;

    void showManager(){
        cout<<"Bonus: "<<bonus<<endl;
        cout<<"Total Salary: "<<salary+bonus<<endl;
    }
};

int main(){
    Manager m;

    m.id=1;
    m.salary=50000;
    m.bonus=10000;

    m.showEmployee();
    m.showManager();
}
