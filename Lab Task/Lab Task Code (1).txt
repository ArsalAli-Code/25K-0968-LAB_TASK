#include <iostream>
using namespace std;

class Person{
public:
    string name;
    int age;

    void displayInfo(){
        cout<<"Name: "<<name<<endl;
        cout<<"Age: "<<age<<endl;
    }
};

class Student : public Person{
public:
    int rollNumber;

    void displayStudent(){
        cout<<"Roll Number: "<<rollNumber<<endl;
    }
};

int main(){
    Student s;

    s.name="Ali";
    s.age=20;
    s.rollNumber=101;

    s.displayInfo();
    s.displayStudent();
}
