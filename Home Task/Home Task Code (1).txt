#include <iostream>
using namespace std;

class Vehicle{
public:
    string brand;
    int speed;

    void showVehicle(){
        cout<<"Brand: "<<brand<<endl;
        cout<<"Speed: "<<speed<<endl;
    }
};

class Car : public Vehicle{
public:
    string fuelType;

    void showCarDetails(){
        cout<<"Fuel Type: "<<fuelType<<endl;
    }
};

int main(){
    Car c;

    c.brand="Toyota";
    c.speed=180;
    c.fuelType="Petrol";

    c.showVehicle();
    c.showCarDetails();
}
